# ---- File 1: motor_sensors.py (left + right publishers) ----
cat > ~/ws4/src/drone_temp_monitor/drone_temp_monitor/motor_sensors.py << 'EOF'
import rclpy
from rclpy.node import Node
from std_msgs.msg import String


class MotorSensor(Node):
    def __init__(self, node_name, label, initial_temp):
        super().__init__(node_name)
        self.label = label
        self.temp = initial_temp
        self.publisher_ = self.create_publisher(String, '/motor_temp', 10)
        self.timer = self.create_timer(3.0, self.timer_callback)  # every 3 seconds

    def timer_callback(self):
        msg = String()
        msg.data = f"{self.label}: {self.temp:.1f}C"
        self.publisher_.publish(msg)
        self.get_logger().info(f"Published: {msg.data}")
        self.temp += 0.5  # +0.5 C every 3 seconds


def main_left(args=None):
    rclpy.init(args=args)
    node = MotorSensor('motor_sensor_left', 'Left Motor', 30.0)    # starts at 30
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


def main_right(args=None):
    rclpy.init(args=args)
    node = MotorSensor('motor_sensor_right', 'Right Motor', 35.0)  # starts at 35
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()
EOF

# ---- File 2: temp_monitoring_node.py (subscriber) ----
cat > ~/ws4/src/drone_temp_monitor/drone_temp_monitor/temp_monitoring_node.py << 'EOF'
import rclpy
from rclpy.node import Node
from std_msgs.msg import String


class TempMonitoringNode(Node):
    def __init__(self):
        super().__init__('temp_monitoring_node')
        self.subscription = self.create_subscription(
            String, '/motor_temp', self.listener_callback, 10)
        self.latest = {'Left Motor': None, 'Right Motor': None}

    def listener_callback(self, msg):
        try:
            motor, temp = msg.data.split(':', 1)
            self.latest[motor.strip()] = temp.strip()
        except ValueError:
            self.get_logger().warn(f"Malformed message: {msg.data}")
            return

        left = self.latest['Left Motor'] or 'N/A'
        right = self.latest['Right Motor'] or 'N/A'
        self.get_logger().info(
            f"\n===== Drone Motor Status Report =====\n"
            f"  Received    : {msg.data}\n"
            f"  Left Motor  -> {left}\n"
            f"  Right Motor -> {right}\n"
            f"====================================="
        )


def main(args=None):
    rclpy.init(args=args)
    node = TempMonitoringNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()
EOF


---------------------------------------------------------------------
 PART 3: EDIT setup.py  (manual step - IMPORTANT, before build)
---------------------------------------------------------------------
Open the file:
    nano ~/ws4/src/drone_temp_monitor/setup.py

Replace the entry_points section with:

    entry_points={
        'console_scripts': [
            'motor_sensor_left = drone_temp_monitor.motor_sensors:main_left',
            'motor_sensor_right = drone_temp_monitor.motor_sensors:main_right',
            'temp_monitoring_node = drone_temp_monitor.temp_monitoring_node:main',
        ],
    },

Save: Ctrl+O, Enter, then exit: Ctrl+X


---------------------------------------------------------------------
 PART 4: BUILD
---------------------------------------------------------------------
cd ~/ws4
colcon build --packages-select drone_temp_monitor
source install/setup.bash

---------------------------------------------------------------------
 NOTES
---------------------------------------------------------------------
- Message type used: std_msgs/msg/String  (carries motor ID + temperature,
  e.g. "Left Motor: 30.5C")
- Left starts at 30C, Right starts at 35C, both +0.5C every 3 seconds.
- If "Package not found": you forgot `source ~/ws4/install/setup.bash`
  or you built from the wrong folder (must be ~/ws4).
- In rqt_graph: set the view to "Nodes/Topics (all)" and press Refresh.
